
# Collect all your .cpp files
file(GLOB SOURCES "*.cpp")

# Create the executable
add_executable(EpidemiologySim ${SOURCES})
target_include_directories(EpidemiologySim PRIVATE "../include")

# Because of FetchContent, "raylib" is now a valid target
target_link_libraries(EpidemiologySim PRIVATE raylib)


# Installation
install(
    TARGETS EpidemiologySim
    RUNTIME DESTINATION bin
)

# On Linux, you might need extra system dependencies
if(UNIX AND NOT APPLE)  
    target_link_libraries(EpidemiologySim PRIVATE m pthread dl rt X11)
endif()
